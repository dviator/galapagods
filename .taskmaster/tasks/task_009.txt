function App() {
  return (
    <GameStateProvider>
      <PhaseProvider>
        <div className="flex h-screen">
          <PhaseTracker />
          <main className="flex-1 ml-48">
            <PhaseRouter />
          </main>
        </div>
      </PhaseProvider>
    </GameStateProvider>
  );
}

const PhaseRouter = () => {
  const { currentPhase } = usePhase();
  switch(currentPhase) {
    case 'combat': return <Combat />;
    case 'shop': return <Shop />;
    case 'death': return <Death />;
    case 'lab': return <Lab />;
  }
};
